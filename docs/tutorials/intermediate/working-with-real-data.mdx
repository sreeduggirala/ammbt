# Working with Real Data

While synthetic data is great for testing ideas, real historical swap data provides ground truth for strategy validation. This tutorial covers loading and using real data from various sources.

## Data Requirements

ammBT expects swap data with these columns:

| Column | Type | Description |
|--------|------|-------------|
| `amount0` | float | Token0 amount (positive = into pool) |
| `amount1` | float | Token1 amount (positive = into pool) |
| `price` | float | Price after swap (token1/token0) |
| `timestamp` | int | Unix timestamp |

Optional but helpful:
- `volume` - Swap volume in USD
- `txn_hash` - Transaction identifier
- `sender` - Address that initiated swap

## Loading from The Graph

### Uniswap V3 Subgraph

```python
import pandas as pd
import requests

def load_v3_swaps_from_subgraph(
    pool_address: str,
    start_timestamp: int,
    end_timestamp: int,
    first: int = 1000,
) -> pd.DataFrame:
    """Load V3 swap data from The Graph."""

    endpoint = "https://api.thegraph.com/subgraphs/name/uniswap/uniswap-v3"

    query = """
    query getSwaps($pool: String!, $startTime: Int!, $endTime: Int!, $first: Int!) {
        swaps(
            where: {
                pool: $pool
                timestamp_gte: $startTime
                timestamp_lte: $endTime
            }
            orderBy: timestamp
            orderDirection: asc
            first: $first
        ) {
            amount0
            amount1
            sqrtPriceX96
            timestamp
            transaction { id }
        }
    }
    """

    variables = {
        "pool": pool_address.lower(),
        "startTime": start_timestamp,
        "endTime": end_timestamp,
        "first": first,
    }

    response = requests.post(
        endpoint,
        json={"query": query, "variables": variables}
    )

    data = response.json()["data"]["swaps"]

    # Convert to DataFrame
    df = pd.DataFrame(data)
    df['amount0'] = df['amount0'].astype(float)
    df['amount1'] = df['amount1'].astype(float)
    df['timestamp'] = df['timestamp'].astype(int)

    # Calculate price from sqrtPriceX96
    df['sqrtPriceX96'] = df['sqrtPriceX96'].astype(float)
    df['price'] = (df['sqrtPriceX96'] / (2**96)) ** 2

    return df[['amount0', 'amount1', 'price', 'timestamp']]

# Example usage
pool = "0x8ad599c3a0ff1de082011efddc58f1908eb6e6d8"  # USDC-ETH 0.3%
swaps = load_v3_swaps_from_subgraph(
    pool_address=pool,
    start_timestamp=1700000000,
    end_timestamp=1700100000,
)
print(f"Loaded {len(swaps)} swaps")
```

### Uniswap V2 Subgraph

```python
def load_v2_swaps_from_subgraph(
    pair_address: str,
    start_timestamp: int,
    end_timestamp: int,
    first: int = 1000,
) -> pd.DataFrame:
    """Load V2 swap data from The Graph."""

    endpoint = "https://api.thegraph.com/subgraphs/name/uniswap/uniswap-v2"

    query = """
    query getSwaps($pair: String!, $startTime: Int!, $endTime: Int!, $first: Int!) {
        swaps(
            where: {
                pair: $pair
                timestamp_gte: $startTime
                timestamp_lte: $endTime
            }
            orderBy: timestamp
            orderDirection: asc
            first: $first
        ) {
            amount0In
            amount0Out
            amount1In
            amount1Out
            timestamp
        }
    }
    """

    variables = {
        "pair": pair_address.lower(),
        "startTime": start_timestamp,
        "endTime": end_timestamp,
        "first": first,
    }

    response = requests.post(
        endpoint,
        json={"query": query, "variables": variables}
    )

    data = response.json()["data"]["swaps"]
    df = pd.DataFrame(data)

    # Convert to standard format
    df['amount0'] = df['amount0In'].astype(float) - df['amount0Out'].astype(float)
    df['amount1'] = df['amount1In'].astype(float) - df['amount1Out'].astype(float)
    df['timestamp'] = df['timestamp'].astype(int)

    # Calculate price
    df['price'] = abs(df['amount1']) / abs(df['amount0'])

    return df[['amount0', 'amount1', 'price', 'timestamp']]
```

## Loading from Dune Analytics

```python
from dune_client.client import DuneClient
from dune_client.query import Query

def load_from_dune(query_id: int, api_key: str) -> pd.DataFrame:
    """Load swap data from a Dune query."""

    dune = DuneClient(api_key)
    query = Query(query_id=query_id)

    result = dune.execute(query)
    df = pd.DataFrame(result.get_rows())

    # Ensure required columns
    required = ['amount0', 'amount1', 'price', 'timestamp']
    assert all(col in df.columns for col in required)

    return df[required]
```

## Data Validation

Before using real data, validate it:

```python
def validate_swap_data(df: pd.DataFrame) -> bool:
    """Validate swap data format and quality."""

    # Check required columns
    required = ['amount0', 'amount1', 'price', 'timestamp']
    missing = [col for col in required if col not in df.columns]
    if missing:
        print(f"Missing columns: {missing}")
        return False

    # Check for nulls
    null_counts = df[required].isnull().sum()
    if null_counts.any():
        print(f"Null values found:\n{null_counts}")
        return False

    # Check types
    if not pd.api.types.is_numeric_dtype(df['amount0']):
        print("amount0 must be numeric")
        return False

    # Check price sanity
    if (df['price'] <= 0).any():
        print(f"Found {(df['price'] <= 0).sum()} non-positive prices")
        return False

    # Check timestamp ordering
    if not df['timestamp'].is_monotonic_increasing:
        print("Timestamps not in order")
        return False

    # Check for duplicates
    dupes = df['timestamp'].duplicated().sum()
    if dupes > 0:
        print(f"Found {dupes} duplicate timestamps")

    print(f"Validation passed!")
    print(f"  Swaps: {len(df)}")
    print(f"  Date range: {pd.to_datetime(df['timestamp'].min(), unit='s')} to {pd.to_datetime(df['timestamp'].max(), unit='s')}")
    print(f"  Price range: {df['price'].min():.6f} - {df['price'].max():.6f}")

    return True

# Usage
is_valid = validate_swap_data(swaps)
```

## Data Preprocessing

### Handling Token Decimals

Different tokens have different decimals:

```python
def adjust_for_decimals(
    df: pd.DataFrame,
    token0_decimals: int,
    token1_decimals: int,
) -> pd.DataFrame:
    """Adjust amounts for token decimals."""

    df = df.copy()
    df['amount0'] = df['amount0'] / (10 ** token0_decimals)
    df['amount1'] = df['amount1'] / (10 ** token1_decimals)

    # Recalculate price in human-readable units
    df['price'] = abs(df['amount1']) / abs(df['amount0'])

    return df

# Example: ETH (18 decimals) / USDC (6 decimals)
swaps = adjust_for_decimals(swaps, token0_decimals=18, token1_decimals=6)
```

### Filtering Outliers

```python
def filter_outliers(df: pd.DataFrame, price_std_threshold: float = 5) -> pd.DataFrame:
    """Remove swaps with outlier prices."""

    mean_price = df['price'].mean()
    std_price = df['price'].std()

    lower = mean_price - price_std_threshold * std_price
    upper = mean_price + price_std_threshold * std_price

    filtered = df[(df['price'] >= lower) & (df['price'] <= upper)]

    removed = len(df) - len(filtered)
    if removed > 0:
        print(f"Removed {removed} outlier swaps ({removed/len(df):.1%})")

    return filtered

swaps = filter_outliers(swaps)
```

### Resampling

If you have too much data, resample:

```python
def resample_swaps(df: pd.DataFrame, interval_seconds: int = 60) -> pd.DataFrame:
    """Aggregate swaps into time buckets."""

    df = df.copy()
    df['bucket'] = (df['timestamp'] // interval_seconds) * interval_seconds

    resampled = df.groupby('bucket').agg({
        'amount0': 'sum',
        'amount1': 'sum',
        'price': 'last',  # Use last price in bucket
        'timestamp': 'last',
    }).reset_index(drop=True)

    return resampled

# Resample to 1-minute buckets
swaps = resample_swaps(swaps, interval_seconds=60)
```

## Running Backtest with Real Data

```python
import ammbt as amm

# Load and validate data
swaps = load_v3_swaps_from_subgraph(
    pool_address="0x8ad599c3a0ff1de082011efddc58f1908eb6e6d8",
    start_timestamp=1700000000,
    end_timestamp=1700500000,
)
swaps = adjust_for_decimals(swaps, 6, 18)  # USDC/ETH
swaps = filter_outliers(swaps)

validate_swap_data(swaps)

# Get initial price for backtester
initial_price = swaps['price'].iloc[0]
print(f"Initial price: {initial_price:.2f}")

# Run backtest
bt = amm.LPBacktester(
    amm_type='v3',
    initial_price=initial_price,
    fee_tier=3000,
)

strategies = {
    'initial_capital': [10000, 25000, 50000],
    'tick_lower': [-200, -400, -600],
    'tick_upper': [200, 400, 600],
    'rebalance_threshold': [0.0, 0.05, 0.10],
}

result = bt.run(swaps, strategies)
print(result.summary().head(10))
```

## Common Issues

<Warning>
**Stale data**: Ensure your data source is up-to-date
</Warning>

<Warning>
**Decimal errors**: Wrong decimals can cause 1000x price errors
</Warning>

<Warning>
**Survivorship bias**: Pools that failed won't be in historical data
</Warning>

<Warning>
**Rate limits**: API calls may be rate-limited; cache data locally
</Warning>
